JavaScript

(function () {
    'use strict';

    var body = document.body;
    var lastTrigger = null;

    function activeInformationMode() {
        return document.documentElement.classList.contains('ccp-information-mode') ||
            (body && body.classList.contains('ccp-information-mode'));
    }

    function removeBackdrop() {
        var backdrop = document.querySelector('.ccp-mobile-menu-backdrop');
        if (backdrop && backdrop.parentNode) {
            backdrop.parentNode.removeChild(backdrop);
        }
    }

    function closePanels(restoreFocus) {
        var panels = document.querySelectorAll('#ccp-mobile-menu, #ccp-info-menu');
        var i;

        for (i = 0; i < panels.length; i++) {
            panels[i].classList.remove('show');
            panels[i].setAttribute('aria-hidden', 'true');
            panels[i].setAttribute('inert', '');
        }

        var toggles = document.querySelectorAll('[data-ccp-mobile-menu-toggle], [data-ccp-info-menu-toggle]');
        for (i = 0; i < toggles.length; i++) {
            toggles[i].setAttribute('aria-expanded', 'false');
        }

        removeBackdrop();

        if (body) {
            body.classList.remove('ccp-mobile-menu-open', 'ccp-info-menu-open');
        }

        if (restoreFocus !== false && lastTrigger && document.documentElement.contains(lastTrigger)) {
            lastTrigger.focus();
        }

        lastTrigger = null;
    }

    function openPanel(selector, type, trigger) {
        var panel = document.querySelector(selector);
        if (!panel || !body) {
            return;
        }

        closePanels(false);
        lastTrigger = trigger || null;

        var backdrop = document.createElement('div');
        backdrop.className = 'ccp-mobile-menu-backdrop';
        backdrop.setAttribute('aria-hidden', 'true');
        backdrop.addEventListener('click', function () {
            closePanels(true);
        });
        body.appendChild(backdrop);

        panel.removeAttribute('inert');
        panel.setAttribute('aria-hidden', 'false');

        if (lastTrigger) {
            lastTrigger.setAttribute('aria-expanded', 'true');
        }

        body.classList.add('ccp-mobile-menu-open');
        if (type === 'info') {
            body.classList.add('ccp-info-menu-open');
        }

        window.requestAnimationFrame(function () {
            panel.classList.add('show');
            var focus = panel.querySelector('[data-ccp-mobile-menu-close], [data-ccp-info-menu-close], a, button');
            if (focus) {
                focus.focus();
            }
        });
    }

    if (body) {
        document.documentElement.classList.add('ccp-information-mode');
        body.classList.add('ccp-information-mode');
    }

    window.CodeCartProTheme = window.CodeCartProTheme || {};
    window.CodeCartProTheme.informationMode = 1;
    window.CodeCartProTheme.quantitySwitch = 0;
    window.CodeCartProTheme.cartDrawer = 0;
    window.CodeCartProTheme.cartToast = 0;
    window.CodeCartProTheme.cartAddPopup = 0;
    window.CodeCartProTheme.directCheckoutAfterAdd = 0;

    document.addEventListener('click', function (event) {
        var target = event.target;
        if (!target || !target.closest) {
            return;
        }

        var catalogToggle = target.closest('[data-ccp-mobile-menu-toggle]');
        if (catalogToggle) {
            event.preventDefault();
            openPanel('#ccp-mobile-menu', 'catalog', catalogToggle);
            return;
        }

        var infoToggle = target.closest('[data-ccp-info-menu-toggle]');
        if (infoToggle) {
            event.preventDefault();
            openPanel('#ccp-info-menu', 'info', infoToggle);
            return;
        }

        if (target.closest('[data-ccp-mobile-menu-close], [data-ccp-info-menu-close]')) {
            event.preventDefault();
            closePanels(true);
            return;
        }

        if (!activeInformationMode()) {
            return;
        }

        var blocked = target.closest(
            '#button-cart, .ccp-card-cart, .ccp-js-quick-order, [data-ccp-mobile-cart], ' +
            'a[href*="route=checkout/cart"], a[href*="route=checkout/checkout"], ' +
            'a[href$="/cart/"], a[href$="/checkout/"]'
        );

        if (blocked) {
            event.preventDefault();
            event.stopPropagation();
            if (event.stopImmediatePropagation) {
                event.stopImmediatePropagation();
            }
        }
    }, true);

    document.addEventListener('keydown', function (event) {
        if (event.key === 'Escape' || event.keyCode === 27) {
            closePanels(true);
        }
    });
})();
